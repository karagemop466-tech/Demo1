Official request-signing contract (documented only — do not run without your own key)

// Signs timestamp + method + path (path includes /trade-api/v2, no query string)
ts = String(Date.now())
message = ts + "POST" + "/trade-api/v2/portfolio/events/orders"
signature = base64( privateKeyEd25519.sign(message) )   // or RSA-PSS/SHA-256
headers = {
  "KALSHI-ACCESS-KEY": "",
  "KALSHI-ACCESS-TIMESTAMP": ts,
  "KALSHI-ACCESS-SIGNATURE": signature,
  "Content-Type": "application/json"
}
body = {
  "ticker": "",
  "side": "bid",                 // bid = buy YES, ask = sell YES
  "count": "1",
  "price": "0.0100",             // dollars; must fit the market's price step
  "time_in_force": "good_till_canceled",
  "self_trade_prevention_type": "taker_at_cross",
  "client_order_id": ""
}
